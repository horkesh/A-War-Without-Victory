const sha256 = (bytes) => crypto.createHash('sha256').update(bytes).digest('hex');
const readJson = (file) => JSON.parse(fs.readFileSync(file, 'utf8'));

function parsePorcelainV1(text) {
  return text.replace(/\r\n/g, '\n').replace(/\n$/, '').split('\n').filter((line) => line.length > 0).map((raw) => {
    if (raw.length < 4 || raw[2] !== ' ') throw new Error(`malformed porcelain-v1 row: ${JSON.stringify(raw)}`);
    return { index: raw[0], worktree: raw[1], path: raw.slice(3), raw };
  });
}

function assessManifestOnlyStatus(text) {
  let rows;
  try { rows = parsePorcelainV1(text); } catch (error) { return { pass: false, rows: [], error: String(error.message) }; }
  const pass = rows.length === 1 && rows[0].index === ' ' && rows[0].worktree === 'M' && rows[0].path === MANIFEST_REL;
  return { pass, rows, error: null };
}

function proveRawManifestHashReplacement(oldBytes, currentBytes, before, after) {
  const oldText = oldBytes.toString('utf8');
  let reverted = currentBytes.toString('utf8');
  const rows = ARTIFACTS.map((name) => {
    const oldHash = before[name];
    const newHash = after[name];
    const oldOccurrences = oldText.split(oldHash).length - 1;
    const newOccurrences = reverted.split(newHash).length - 1;
    const pass = oldHash !== newHash && oldOccurrences === 1 && newOccurrences === 1;
    if (pass) reverted = reverted.replace(newHash, oldHash);
    return { name, old_hash: oldHash, new_hash: newHash, old_occurrences: oldOccurrences, new_occurrences: newOccurrences, pass };
  });
  const reconstructed = Buffer.from(reverted, 'utf8');
  return { pass: rows.every((row) => row.pass) && reconstructed.equals(oldBytes), rows, reconstructed_sha256: sha256(reconstructed), old_sha256: sha256(oldBytes), bytes_exact_after_replacing_eight_hashes: reconstructed.equals(oldBytes) };
}

function arg(argv, name) {

}

function compactInventory(inventoryPath, appendPaths = [], excludePaths = []) {
  const inventory = readJson(inventoryPath);
  if (!Array.isArray(inventory.files)) throw new Error('preservation inventory files missing');
  const append = new Set(appendPaths.map((p) => path.resolve(ROOT, p).toLowerCase()));
  const exclude = new Set(excludePaths.map((p) => path.resolve(ROOT, p).toLowerCase()));
  const cache = new Map();
  const failures = [];
  const telemetry = [];
  let checked = 0;
  for (const expected of inventory.files) {
    const absolute = path.resolve(ROOT, expected.path);
    if (exclude.has(absolute.toLowerCase())) continue;
    checked += 1;
    if (!fs.existsSync(absolute)) { failures.push({ path: expected.path, reason: 'missing' }); continue; }
    const key = `${absolute.toLowerCase()}\0${expected.bytes}\0${append.has(absolute.toLowerCase()) ? 'prefix' : 'exact'}`;
    let actual = cache.get(key);
    if (!actual) {
      const bytes = fs.readFileSync(absolute);
      actual = append.has(absolute.toLowerCase())
        ? { mode: 'append_prefix', current_bytes: bytes.length, opening_prefix_sha256: sha256(bytes.subarray(0, expected.bytes)), append_bytes: bytes.length - expected.bytes }
        : { mode: 'exact', current_bytes: bytes.length, current_sha256: sha256(bytes) };
      cache.set(key, actual);
    }
    const pass = actual.mode === 'append_prefix'
      ? actual.current_bytes >= expected.bytes && actual.opening_prefix_sha256 === expected.sha256
      : actual.current_bytes === expected.bytes && actual.current_sha256 === expected.sha256;
    if (!pass) failures.push({ path: expected.path, expected_bytes: expected.bytes, expected_sha256: expected.sha256, ...actual });
    if (actual.mode === 'append_prefix') telemetry.push({ path: expected.path, expected_bytes: expected.bytes, expected_sha256: expected.sha256, ...actual, pass });
  }
  return { expected_count: inventory.files.length, checked_count: checked, excluded_count: inventory.files.length - checked, unique_files_hashed: cache.size, telemetry, failures, pass: failures.length === 0 };
}

function verifySnapshot(snapshotPath, exclude = []) {
  const snapshot = readJson(snapshotPath);
  const excluded = new Set(exclude);
  const failures = [];
  let checked = 0;
  for (const expected of snapshot.files ?? []) {
    if (excluded.has(expected.path)) continue;
    checked += 1;
    const absolute = path.join(ROOT, expected.path);
    if (!fs.existsSync(absolute)) { failures.push({ path: expected.path, reason: 'missing' }); continue; }
    const bytes = fs.readFileSync(absolute);
    const actualSha = sha256(bytes);
    if (bytes.length !== expected.bytes || actualSha !== expected.sha256) failures.push({ path: expected.path, expected_bytes: expected.bytes, actual_bytes: bytes.length, expected_sha256: expected.sha256, actual_sha256: actualSha });
  }
  return { expected_count: snapshot.files?.length ?? 0, checked_count: checked, excluded_count: (snapshot.files?.length ?? 0) - checked, failures, pass: failures.length === 0 };
}
