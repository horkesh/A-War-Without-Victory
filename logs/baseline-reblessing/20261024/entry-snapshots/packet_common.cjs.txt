const sha256 = (bytes) => crypto.createHash('sha256').update(bytes).digest('hex');

function listFilesRecursively(root) {
  if (!fs.existsSync(root)) return [];
  const files = [];
  const visit = (dir) => {
    const entries = fs.readdirSync(dir, { withFileTypes: true }).sort((a, b) => a.name < b.name ? -1 : a.name > b.name ? 1 : 0);
    for (const entry of entries) {
      const absolute = path.join(dir, entry.name);
      if (entry.isDirectory()) visit(absolute);
      else if (entry.isFile()) files.push(absolute);
      else throw new Error(`unsupported filesystem entry in guarded tree: ${absolute}`);
    }
  };
  visit(root);
  return files;
}

function assertNoFilesRecursively(root) {
  const files = listFilesRecursively(root);
  if (files.length > 0) {
    throw new Error(`refusing to overwrite existing file(s) under ${root}: ${files.join(', ')}`);
  }
  return files;
}

function fileRecord(root, absolute) {
  const bytes = fs.readFileSync(absolute);
  return {
    path: path.relative(root, absolute).replaceAll('\\', '/'),
    bytes: bytes.length,
    sha256: sha256(bytes),
  };
}

function snapshotTree(root, target) {
  const files = listFilesRecursively(target).map((absolute) => fileRecord(root, absolute));
  return {
    target: path.relative(root, target).replaceAll('\\', '/'),
    file_count: files.length,
    aggregate_sha256: sha256(Buffer.from(files.map((row) => `${row.path}\0${row.bytes}\0${row.sha256}\n`).join(''), 'utf8')),
    files,
  };
}

function parseAuthorizedArgs(argv) {
  const evidenceDir = argv[0];
  const headAt = argv.indexOf('--expected-head');
  const expectedHead = headAt >= 0 ? argv[headAt + 1] : undefined;
  if (!evidenceDir) throw new Error('evidence directory is required');
  if (!/^[0-9a-f]{40}$/i.test(expectedHead ?? '')) throw new Error('--expected-head requires a 40-character hexadecimal commit');
  if (!argv.includes('--owner-authorized-once')) throw new Error('owner authorization token required');
  return { evidenceDir, expectedHead: expectedHead.toLowerCase() };
}

function scrubNotificationAndOverrideKeys(env) {
  const next = { ...env };
  for (const key of Object.keys(next)) {
    if (/slack|webhook/i.test(key)) delete next[key];
  }
  delete next.AWWV_PROVENANCE_OVERRIDE;
  return next;
}

function buildCampaignEnvironment(base) {
  const env = scrubNotificationAndOverrideKeys(base);
  delete env.UPDATE_BASELINES;
  delete env.AWWV_PLAYTEST_TELEMETRY;
  delete env.C_LANE_CORPS_DIRECTIVE_TELEMETRY_DISABLED;
  env.AWWV_S6_GRADE_RUN = 'true';
  env.AWWV_DEBUG_AXIS_READINESS = 'Mistral';
  env.AWWV_MISTRAL_PREP_TRACE = '1';
  env.AWWV_DEBUG_REASON_CODES = 'movement_phase,commander_creation,battle_stack,operation_capture_ownership,mistral_tg_activation';
  env.AWWV_DEBUG_MOVEMENT_FORMATIONS = 'hvo_1st_guard_abb';
  return env;
}

function buildUpdateEnvironment(base) {
  const env = scrubNotificationAndOverrideKeys(base);
  for (const key of Object.keys(env)) {
    if (/^AWWV_DEBUG_/.test(key) || key === 'AWWV_MISTRAL_PREP_TRACE' || key === 'AWWV_PLAYTEST_TELEMETRY') delete env[key];
  }
  env.AWWV_S6_GRADE_RUN = 'true';
  delete env.C_LANE_CORPS_DIRECTIVE_TELEMETRY_DISABLED;
  env.UPDATE_BASELINES = '1';
  return env;
}

function projectDefaultWeeklyReport(input) {
  const text = input.toString('utf8');
  const rows = text.trimEnd().split('\n').filter(Boolean).map(JSON.parse);
  const removed = {
    movement_phase_diagnostics_rows: 0,
    attacker_brigades_rows: 0,
    defender_contributions_rows: 0,
  };
  for (const row of rows) {
    if (Object.hasOwn(row, 'movement_phase_diagnostics')) {
      removed.movement_phase_diagnostics_rows += 1;
      delete row.movement_phase_diagnostics;
    }
    for (const battle of row.battles ?? []) {
      if (Object.hasOwn(battle, 'attacker_brigades')) {
        removed.attacker_brigades_rows += 1;
        delete battle.attacker_brigades;
      }
      if (Object.hasOwn(battle, 'defender_contributions')) {
        removed.defender_contributions_rows += 1;
        delete battle.defender_contributions;
      }
    }
  }
  const bytes = Buffer.from(rows.map(JSON.stringify).join('\n') + '\n', 'utf8');
  return { bytes, rows: rows.length, removed, sha256: sha256(bytes) };
}

function validateEightHashManifestDelta(beforeHashes, afterHashes) {
  const beforeNames = Object.keys(beforeHashes).sort();
  const afterNames = Object.keys(afterHashes).sort();
  if (JSON.stringify(beforeNames) !== JSON.stringify(BASELINE_ARTIFACTS) || JSON.stringify(afterNames) !== JSON.stringify(BASELINE_ARTIFACTS)) {
    throw new Error('baseline artifact set must remain exactly the canonical eight');
  }
  for (const [name, value] of Object.entries(afterHashes)) {
    if (!/^[0-9a-f]{64}$/.test(value)) throw new Error(`invalid SHA-256 for ${name}`);
  }
  const changed = BASELINE_ARTIFACTS.filter((name) => beforeHashes[name] !== afterHashes[name]);
  if (changed.length !== 8) throw new Error(`expected exactly eight changed artifact hashes, observed ${changed.length}`);
  return { changed };
}

function validateAcceptedCandidateHashes(actualHashes, projectedWeeklyHash) {
  const failures = [];
  for (const name of BASELINE_ARTIFACTS) {
    const actual = name === 'weekly_report.jsonl' ? projectedWeeklyHash : actualHashes[name];
    if (actual !== ACCEPTED_CI_HASHES[name]) failures.push({ artifact: name, expected: ACCEPTED_CI_HASHES[name], actual: actual ?? null });
  }
  return { pass: failures.length === 0, failures };
}

function verifyPreservationInventory(inventoryPath, appendOnlyPaths = [], excludedPaths = []) {
  const inventory = JSON.parse(fs.readFileSync(inventoryPath, 'utf8'));
  if (!Array.isArray(inventory.files)) throw new Error(`invalid preservation inventory: ${inventoryPath}`);
  const appendSet = new Set(appendOnlyPaths.map((entry) => path.resolve(entry).toLowerCase()));
  const excludedSet = new Set(excludedPaths.map((entry) => path.resolve(entry).toLowerCase()));
  const rows = inventory.files.filter((expected) => !excludedSet.has(path.resolve(expected.path).toLowerCase())).map((expected) => {
    const absolute = path.resolve(expected.path);
    if (!fs.existsSync(absolute)) return { ...expected, absolute, pass: false, reason: 'missing' };
    const bytes = fs.readFileSync(absolute);
    if (appendSet.has(absolute.toLowerCase())) {
      const prefix = bytes.subarray(0, expected.bytes);
      const pass = bytes.length >= expected.bytes && sha256(prefix) === expected.sha256;
      return { ...expected, absolute, current_bytes: bytes.length, opening_prefix_sha256: sha256(prefix), append_bytes: bytes.length - expected.bytes, pass, mode: 'append_prefix' };
    }
    const currentSha = sha256(bytes);
    return { ...expected, absolute, current_bytes: bytes.length, current_sha256: currentSha, pass: bytes.length === expected.bytes && currentSha === expected.sha256, mode: 'exact' };
  });
  return { source: path.resolve(inventoryPath), expected_count: inventory.files.length, checked_count: rows.length, excluded_count: inventory.files.length - rows.length, pass: rows.every((row) => row.pass), failures: rows.filter((row) => !row.pass), rows };
}

function verifyDependencyInventory(inventoryPath, requiredPaths) {
  const inventory = JSON.parse(fs.readFileSync(inventoryPath, 'utf8'));
  if (!Array.isArray(inventory.files)) throw new Error(`invalid dependency inventory: ${inventoryPath}`);
  const byPath = new Map(inventory.files.map((row) => [path.resolve(row.path).toLowerCase(), row]));
  const rows = requiredPaths.map((required) => {
    const absolute = path.resolve(required);
    const expected = byPath.get(absolute.toLowerCase());
    if (!expected) return { path: absolute, pass: false, reason: 'not listed' };
    if (!fs.existsSync(absolute)) return { ...expected, path: absolute, pass: false, reason: 'missing' };
    const bytes = fs.readFileSync(absolute);
    const currentSha = sha256(bytes);
    return { ...expected, path: absolute, current_bytes: bytes.length, current_sha256: currentSha, pass: bytes.length === expected.bytes && currentSha === expected.sha256 };
  });
  return { source: path.resolve(inventoryPath), required_count: requiredPaths.length, pass: rows.every((row) => row.pass), failures: rows.filter((row) => !row.pass), rows };
}
