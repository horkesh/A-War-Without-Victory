**Goal:** Wire up the existing AI commander infrastructure (P0), then add three narrative features that make AI commanders produce player-facing content: officers who talk back (P1), after-action narrative (P2), and war dispatches (P3).

**Architecture:** All four features share the same `src/sim/ai_commander/` module. P0 connects the existing code to Electron IPC so desktop play works. New pipeline steps gated by `ai_commander_config.mode !== 'cadet'`.

**Tech Stack:** TypeScript, Vitest, `@anthropic-ai/sdk` (already in package.json), Electron IPC, React.

**Cost model:** P1 ~$0.01/turn, P2 ~$0.005/battle, P3 ~$0.005/turn. Total ~$2-3/game at Officer tier.

// src/sim/ai_commander/corps_dialogue.ts
/**
 * Corps Dialogue — officers respond to orders in character.
 * Haiku call per active corps per turn. ~$0.01/turn total.
 */
import type { AiPrompt, AiResponse } from './ai_types.js';
import type { AiClient } from './ai_client.js';
import type { FactionId, GameState } from '../../state/game_state.js';
import { getCorpsCommanderProfile } from './personality_profiles.js';
import { MODEL_ROUTING, AI_TEMPERATURE } from './ai_config.js';
import type { AiCommanderMode } from './ai_config.js';
import { getCorpsCommander } from '../combat/officer_system.js';
import { logDecision } from './decision_log.js';

export interface CorpsDialogueEntry {
    turn: number;
    corps_id: string;
    faction: FactionId;
    officer_name: string;
    acknowledgment: string;
    concern: string | null;
    confidence: 'high' | 'medium' | 'low';
}

export interface DialoguePromptInput {
    officerName: string;
    faction: FactionId;
    corpsId: string;
    competence: number;
    aggressiveness: number;
    defensiveSkill: number;
    stance: string;
    orderContext: string;
    corpsStatus: string;
    recentBattles: string;
}

export function buildCorpsDialoguePrompt(input: DialoguePromptInput): AiPrompt {
    const system = `You are ${input.officerName}, a corps commander in the ${
        input.faction === 'RS' ? 'VRS' : input.faction === 'RBiH' ? 'ARBiH' : 'HVO'
    }. Competence: ${input.competence}/5. Aggressiveness: ${input.aggressiveness}/5.

Respond to the order you've received. Be brief and in character.
${input.aggressiveness >= 4 ? 'You are aggressive — you push for action and may express frustration with defensive orders.' : ''}
${input.aggressiveness <= 2 ? 'You are cautious — you may express concern about risky attacks.' : ''}
${input.competence >= 4 ? 'Your analysis is sharp and specific.' : ''}
${input.competence <= 2 ? 'Your analysis may miss important factors.' : ''}

Output MUST be valid JSON:
{
  "acknowledgment": "1-2 sentences acknowledging the order, in your voice",
  "concern": "null if no concern, or 1-2 sentences raising an objection or risk",
  "confidence": "high | medium | low"
}`;

    const user = `CURRENT STANCE: ${input.stance}
ORDER CONTEXT: ${input.orderContext}
CORPS STATUS: ${input.corpsStatus}
RECENT COMBAT: ${input.recentBattles}

Respond in character.`;

    return {
        system,
        user,
        model: 'claude-haiku-4-5-20251001',
        max_tokens: 256,
        temperature: AI_TEMPERATURE,
    };
}

export function parseDialogueResponse(raw: string): Omit<CorpsDialogueEntry, 'turn' | 'corps_id' | 'faction' | 'officer_name'> | null {
    try {
        const stripped = raw.match(/```(?:json)?\s*\n?([\s\S]*?)\n?\s*```/);
        const text = stripped ? stripped[1].trim() : raw.trim();
        const data = JSON.parse(text);
        if (!data.acknowledgment) return null;
        return {
            acknowledgment: String(data.acknowledgment),
            concern: data.concern ? String(data.concern) : null,
            confidence: ['high', 'medium', 'low'].includes(data.confidence) ? data.confidence : 'medium',
        };
    } catch {
        return null;
    }
}

/** Generate dialogue for all active bot corps. Returns entries to store on state. */
export async function generateCorpsDialogues(
    state: GameState,
    client: AiClient | null
): Promise<CorpsDialogueEntry[]> {
    if (!client || !client.isAvailable()) return [];

    const entries: CorpsDialogueEntry[] = [];
    const corpsCommand = state.military.corps_command;
    if (!corpsCommand) return [];

    const formations = state.military.formations ?? {};
    const playerFaction = (state.meta as any).player_faction;
    const turn = state.meta.turn;

    const corpsIds = Object.keys(corpsCommand).sort();
    for (const corpsId of corpsIds) {
        const cmd = corpsCommand[corpsId];
        if (!cmd) continue;
        const corps = formations[corpsId];
        if (!corps || corps.status !== 'active') continue;
        // Skip player faction corps — player doesn't need AI dialogue for own orders
        if (corps.faction === playerFaction) continue;

        const officer = getCorpsCommander(corpsId, state);
        const officerName = officer?.data.name ?? corpsId;
        const competence = officer?.data.competence ?? 3;
        const aggressiveness = officer?.data.aggressiveness ?? 3;
        const defensiveSkill = officer?.data.defensive_skill ?? 3;

        const stance = cmd.directive?.stance_override ?? 'balanced';
        const hasOp = !!cmd.active_operation;
        const opTarget = cmd.active_operation?.objective_osid ?? 'none';

        const prompt = buildCorpsDialoguePrompt({
            officerName,
            faction: corps.faction as FactionId,
            corpsId,
            competence,
            aggressiveness,
            defensiveSkill,
            stance,
            orderContext: hasOp
                ? `Active operation targeting ${opTarget}. Army stance: ${stance}.`
                : `No active operation. Army stance: ${stance}. Hold current positions.`,
            corpsStatus: `${Object.values(formations).filter(f => f.corps_id === corpsId && f.kind === 'brigade' && f.status === 'active').length} active brigades.`,
            recentBattles: `${cmd.active_operation?.battles_this_turn ?? 0} battles this turn. ${cmd.active_operation?.territory_gained_this_turn ?? 0} OSIDs captured.`,
        });

        try {
            const response = await client.generateDecision(prompt);
            const parsed = parseDialogueResponse(response.content);
            if (parsed) {
                entries.push({
                    turn,
                    corps_id: corpsId,
                    faction: corps.faction as FactionId,
                    officer_name: officerName,
                    ...parsed,
                });
            }
        } catch {
            // Silent — dialogue is cosmetic
        }
    }

    return entries;
}
```

// src/sim/ai_commander/aar_narrative.ts
/**
 * After-Action Narrative — AI narrates significant battles in character.
 * Haiku call per significant battle. ~$0.005/battle.
 */
import type { AiPrompt } from './ai_types.js';
import type { AiClient } from './ai_client.js';
import type { FactionId, GameState } from '../../state/game_state.js';
import { AI_TEMPERATURE } from './ai_config.js';

export interface BattleNarrative {
    turn: number;
    target_osid: string;
    corps_id: string;
    faction: FactionId;
    officer_name: string;
    narrative: string;
    tone: string;
    outcome: string;
}

export interface AARPromptInput {
    officerName: string;
    faction: FactionId;
    corpsId: string;
    targetOsid: string;
    outcome: string;
    attackerCasualties: number;
    defenderCasualties: number;
    attackerBrigades: string[];
    defenderBrigades: string[];
    territoryChanged: boolean;
}

export interface NarrativeQueueEntry {
    faction: FactionId;
    corpsId: string;
    input: AARPromptInput;
}

export function isSignificantBattle(info: {
    outcome: string;
    totalCasualties: number;
    territoryChanged: boolean;
}): boolean {
    if (info.outcome === 'decisive' || info.outcome === 'catastrophic') return true;
    if (info.territoryChanged) return true;
    if (info.totalCasualties >= 200) return true;
    return false;
}

export function buildAARPrompt(input: AARPromptInput): AiPrompt {
    const factionName = input.faction === 'RS' ? 'VRS' : input.faction === 'RBiH' ? 'ARBiH' : 'HVO';
    const municipality = input.targetOsid.split(':')[1] ?? 'unknown';
    const won = input.outcome === 'decisive' || input.outcome === 'victory' || input.outcome === 'costly_victory';

    return {
        system: `You are ${input.officerName}, corps commander of the ${factionName}. Write a brief field report about a battle that just occurred. 2-4 sentences, in character. ${won ? 'Your forces prevailed.' : 'Your forces were repulsed or took heavy losses.'}

Output valid JSON: { "narrative": "...", "tone": "triumphant|somber|defiant|grim|matter-of-fact" }`,
        user: `BATTLE REPORT:
Location: ${municipality} (${input.targetOsid})
Outcome: ${input.outcome}
Our casualties: ${input.attackerCasualties}
Enemy casualties: ${input.defenderCasualties}
Our brigades engaged: ${input.attackerBrigades.join(', ')}
Enemy brigades: ${input.defenderBrigades.join(', ')}
Territory changed hands: ${input.territoryChanged ? 'YES' : 'NO'}

Write the field report.`,
        model: 'claude-haiku-4-5-20251001',
        max_tokens: 384,
        temperature: AI_TEMPERATURE,
    };
}

export function parseAARResponse(raw: string): { narrative: string; tone: string } | null {
    try {
        const stripped = raw.match(/```(?:json)?\s*\n?([\s\S]*?)\n?\s*```/);
        const text = stripped ? stripped[1].trim() : raw.trim();
        const data = JSON.parse(text);
        if (!data.narrative) return null;
        return {
            narrative: String(data.narrative),
            tone: String(data.tone ?? 'matter-of-fact'),
        };
    } catch {
        return null;
    }
}

/** Generate narratives for all queued battles. */
export async function generateBattleNarratives(
    state: GameState,
    queue: NarrativeQueueEntry[],
    client: AiClient | null
): Promise<BattleNarrative[]> {
    if (!client || !client.isAvailable() || queue.length === 0) return [];

    const narratives: BattleNarrative[] = [];
    const turn = state.meta.turn;

    // Cap at 5 narratives per turn to control cost
    const capped = queue.slice(0, 5);

    for (const entry of capped) {
        try {
            const prompt = buildAARPrompt(entry.input);
            const response = await client.generateDecision(prompt);
            const parsed = parseAARResponse(response.content);
            if (parsed) {
                narratives.push({
                    turn,
                    target_osid: entry.input.targetOsid,
                    corps_id: entry.corpsId,
                    faction: entry.faction,
                    officer_name: entry.input.officerName,
                    narrative: parsed.narrative,
                    tone: parsed.tone,
                    outcome: entry.input.outcome,
                });
            }
        } catch {
            // Silent — narratives are cosmetic
        }
    }

    return narratives;
}
```

// src/sim/ai_commander/war_dispatches.ts
/**
 * War Dispatches — weekly reports from outside perspectives.
 * UN observer, journalist, refugee. Generated from actual game events.
 * Haiku call every 4 turns. ~$0.005/dispatch.
 */
import type { AiPrompt } from './ai_types.js';
import type { AiClient } from './ai_client.js';
import type { GameState } from '../../state/game_state.js';
import { AI_TEMPERATURE } from './ai_config.js';

export interface WarDispatch {
    turn: number;
    source: string;
    headline: string;
    body: string;
    perspective: 'humanitarian' | 'military' | 'civilian' | 'diplomatic';
}

export interface DispatchPromptInput {
    turn: number;
    startWeek: number;
    year: number;
    month: string;
    totalDisplaced: number;
    recentDisplaced: number;
    recentKilled: number;
    siegedCities: string[];
    recentBattles: number;
    territoryChanges: number;
    factionTerritoryPct: Record<string, number>;
}

const PERSPECTIVES = ['humanitarian', 'military', 'civilian', 'diplomatic'] as const;

/** Generate a dispatch every 4 turns (monthly). */
export function shouldGenerateDispatch(turn: number): boolean {
    return turn % 4 === 0;
}

export function buildDispatchPrompt(input: DispatchPromptInput): AiPrompt {
    // Rotate perspective based on turn
    const perspective = PERSPECTIVES[Math.floor(input.turn / 4) % PERSPECTIVES.length];

    const sourceMap = {
        humanitarian: 'You are a UNHCR field officer writing a situation report.',
        military: 'You are a foreign military analyst writing an intelligence brief for NATO.',
        civilian: 'You are a Bosnian civilian writing to a relative abroad.',
        diplomatic: 'You are a UN mediator writing a confidential assessment for the Security Council.',
    };

    return {
        system: `${sourceMap[perspective]}

The year is ${input.year}, month of ${input.month}. The Bosnian War is ongoing.
Write a brief dispatch (3-5 sentences) based on the data below. Be factual about the numbers but convey the human reality. Reference real organizations (UNHCR, ICRC, BBC, UNPROFOR) and use period-appropriate language.

Output valid JSON:
{
  "source": "e.g. UNHCR Field Report, BBC World Service, Private letter",
  "headline": "Brief headline (10 words max)",
  "body": "3-5 sentence dispatch",
  "perspective": "${perspective}"
}`,
        user: `WAR DATA — Week ${input.turn}:
Total displaced: ${input.totalDisplaced.toLocaleString()}
Displaced this month: ${input.recentDisplaced.toLocaleString()}
Killed this month: ${input.recentKilled.toLocaleString()}
Cities under siege: ${input.siegedCities.join(', ') || 'None'}
Battles this month: ${input.recentBattles}
Territory changes: ${input.territoryChanges}
Territory control: RS ${(input.factionTerritoryPct.RS * 100).toFixed(0)}% / RBiH ${(input.factionTerritoryPct.RBiH * 100).toFixed(0)}% / HRHB ${(input.factionTerritoryPct.HRHB * 100).toFixed(0)}%

Write the dispatch.`,
        model: 'claude-haiku-4-5-20251001',
        max_tokens: 384,
        temperature: AI_TEMPERATURE,
    };
}

export function parseDispatchResponse(raw: string): WarDispatch | null {
    try {
        const stripped = raw.match(/```(?:json)?\s*\n?([\s\S]*?)\n?\s*```/);
        const text = stripped ? stripped[1].trim() : raw.trim();
        const data = JSON.parse(text);
        if (!data.body || !data.headline) return null;
        return {
            turn: 0, // Set by caller
            source: String(data.source ?? 'Field Report'),
            headline: String(data.headline),
            body: String(data.body),
            perspective: PERSPECTIVES.includes(data.perspective) ? data.perspective : 'humanitarian',
        };
    } catch {
        return null;
    }
}

/** Gather humanitarian data from game state and generate dispatch. */
export async function generateWarDispatch(
    state: GameState,
    client: AiClient | null
): Promise<WarDispatch | null> {
    if (!client || !client.isAvailable()) return null;

    const turn = state.meta.turn;
    const startWeek = state.meta.war_start_turn ?? 0;
    const weeksSinceStart = turn - startWeek;
    const year = 1992 + Math.floor(weeksSinceStart / 52);
    const monthIndex = Math.floor((weeksSinceStart % 52) / 4.33);
    const months = ['January','February','March','April','May','June','July','August','September','October','November','December'];
    const month = months[monthIndex % 12] ?? 'Unknown';

    // Count displaced
    const displacementLog = state.displacement?.displacement_event_log ?? [];
    const totalDisplaced = displacementLog.reduce((sum, e) => sum + (e.displaced_count ?? 0), 0);
    const recentDisplaced = displacementLog
        .filter(e => e.turn >= turn - 4)
        .reduce((sum, e) => sum + (e.displaced_count ?? 0), 0);

    // Count killed (from casualty ledger or control events)
    const controlEvents = state.political?.control_events ?? [];
    const recentBattles = controlEvents.filter(e => e.turn >= turn - 4 && e.mechanism === 'combat').length;
    const territoryChanges = controlEvents.filter(e => e.turn >= turn - 4).length;

    // Besieged cities
    const siegedCities: string[] = [];
    const enclaveConfig = ['Sarajevo', 'Srebrenica', 'Goražde', 'Žepa', 'Bihać'];
    // Simplified — just check always-besieged
    if (state.military?.enclave_state) {
        for (const name of enclaveConfig) {
            if (state.military.enclave_state[name.toLowerCase()]?.status === 'besieged' ||
                state.military.enclave_state[name.toLowerCase()]?.status === 'strained') {
                siegedCities.push(name);
            }
        }
    }
    if (siegedCities.length === 0) siegedCities.push('Sarajevo'); // Always besieged

    // Territory %
    const pc = state.political?.political_controllers ?? {};
    const total = Object.keys(pc).length || 1;
    const counts: Record<string, number> = { RS: 0, RBiH: 0, HRHB: 0 };
    for (const faction of Object.values(pc)) {
        if (counts[faction as string] !== undefined) counts[faction as string]++;
    }
    const factionTerritoryPct = {
        RS: counts.RS / total,
        RBiH: counts.RBiH / total,
        HRHB: counts.HRHB / total,
    };

    const prompt = buildDispatchPrompt({
        turn,
        startWeek,
        year,
        month,
        totalDisplaced,
        recentDisplaced,
        recentKilled: 0, // Simplified — casualty ledger integration deferred
        siegedCities,
        recentBattles,
        territoryChanges,
        factionTerritoryPct,
    });

    try {
        const response = await client.generateDecision(prompt);
        const parsed = parseDispatchResponse(response.content);
        if (parsed) {
            parsed.turn = turn;
            return parsed;
        }
    } catch {
        // Silent
    }
    return null;
}
